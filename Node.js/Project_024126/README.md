# Node.js: Audit — Baseline Setup for Offline Classroom (Project 024126)

## Project overview

A classroom operating offline provides the setting for this audit exercise. The code is an AI-generated reference implementation based on the student guide and uses fictional data.

## Problem statement

A classroom operating offline needs a repeatable way to handle audit from local records. Case `CLASS-09` illustrates baseline setup.

## Project objectives

- Check whether recorded asset reviews meet a cutoff and list departures.
- Establish a repeatable reference run for one input record.
- Verify the sample result, repeat the run, and observe how bad input is handled.

## Complete Node.js source code

```javascript
// Local reference implementation for project 024126.
import { readFileSync } from "node:fs";
import { pathToFileURL } from "node:url";
import { parseArgs } from "node:util";

function object(value, name) {
  if (typeof value !== "object" || value === null || Array.isArray(value)) {
    throw new Error(`${name} must be an object`);
  }
  return value;
}

function text(value, name, trim = true) {
  if (typeof value !== "string" || !value.trim()) throw new Error(`${name} must be non-empty text`);
  return trim ? value.trim() : value;
}

function day(value, name) {
  const date = text(value, name);
  const epoch = Date.parse(date + "T00:00:00.000Z");
  if (!/^\d{4}-\d{2}-\d{2}$/.test(date) || !Number.isFinite(epoch)
    || new Date(epoch).toISOString().slice(0, 10) !== date) {
    throw new Error(`${name} must be a valid YYYY-MM-DD date`);
  }
  return epoch;
}

function errorMessage(error) {
  return error instanceof Error ? error.message : String(error);
}

function load(path) {
  const value = JSON.parse(readFileSync(path, "utf8"));
  return object(value, "input");
}

function evaluate(record) {
  const reviewed = day(record.reviewed_on, "reviewed_on");
  const cutoff = day(record.cutoff, "cutoff");
  return { owner: text(record.owner, "owner"), reviewed_on: record.reviewed_on,
    follow_up: reviewed < cutoff };
}

function reply(value, status = 0) {
  return { status, output: JSON.stringify(value, null, 2) };
}

export function run(args) {
  const usage = "Usage: node project.mjs sample.json";
  try {
    const { values, positionals } = parseArgs({
      args, allowPositionals: true, options: { help: { type: "boolean" } },
    });
    if (values.help) return { status: 0, output: usage };
    if (positionals.length !== 1) throw new Error(usage);
    const [input] = positionals;

    const record = load(input);
    return reply({ baseline_for: text(record.asset, "asset"), result: evaluate(record) });
  } catch (error) {
    return { status: 2, output: "", error: "Input or output error: " + errorMessage(error) };
  }
}

// Keep the command-line entry point separate from the testable application logic.
if (process.argv[1] && import.meta.url === pathToFileURL(process.argv[1]).href) {
  const result = run(process.argv.slice(2));
  if (result.output) console.log(result.output);
  if (result.error) console.error(result.error);
  process.exitCode = result.status;
}
```

## Implementation explanation

`evaluate()` applies the audit rules to one record. Input JSON supplies one record and the CLI identifies the case so later runs can be compared. The exported `run()` function handles arguments and returns the status and output used by the command-line entry point.

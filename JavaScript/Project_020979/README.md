# JavaScript: Recovery Test — Reporting for Offline Classroom (Project 020979)

## Project overview

A classroom operating offline provides the setting for this recovery test exercise. The code is an AI-generated reference implementation based on the student guide and uses fictional data.

## Problem statement

A classroom operating offline needs a repeatable way to handle recovery test from local records. Case `CLASS-09` illustrates reporting.

## Project objectives

- Restore a mock asset snapshot in memory and compare it against expected values.
- Create a portable JSON report without replacing an existing file.
- Verify the sample result, repeat the run, and observe how bad input is handled.

## Complete JavaScript source code

```javascript
// Local reference implementation for project 020979.
import { readFileSync, writeFileSync } from "node:fs";
import { pathToFileURL } from "node:url";
import { parseArgs, isDeepStrictEqual } from "node:util";

const PROJECT_ID = "020979";

function object(value, name) {
  if (typeof value !== "object" || value === null || Array.isArray(value)) {
    throw new Error(`${name} must be an object`);
  }
  return value;
}

function text(value, name, trim = true) {
  if (typeof value !== "string" || !value.trim()) throw new Error(`${name} must be non-empty text`);
  return trim ? value.trim() : value;
}

function errorMessage(error) {
  return error instanceof Error ? error.message : String(error);
}

function load(path) {
  const value = JSON.parse(readFileSync(path, "utf8"));
  return object(value, "input");
}

function evaluate(record) {
  const restored = structuredClone(object(record.snapshot, "snapshot"));
  const expected = object(record.expected, "expected");
  const keys = new Set([...Object.keys(restored), ...Object.keys(expected)]);
  const differences = [...keys].filter(key => !Object.hasOwn(restored, key)
    || !Object.hasOwn(expected, key) || !isDeepStrictEqual(restored[key], expected[key])).sort();
  return { restored, differences, verified: !differences.length };
}

function reply(value, status = 0) {
  return { status, output: JSON.stringify(value, null, 2) };
}

export function run(args) {
  const usage = "Usage: node project.mjs sample.json [--output report.json]";
  try {
    const { values, positionals } = parseArgs({
      args, allowPositionals: true, options: { output: { type: "string" }, help: { type: "boolean" } },
    });
    if (values.help) return { status: 0, output: usage };
    if (positionals.length !== 1) throw new Error(usage);
    const [input] = positionals;

    const record = load(input);
    const report = reply({ project: PROJECT_ID, asset: text(record.asset, "asset"), result: evaluate(record) });
    if (values.output !== undefined) {
      writeFileSync(values.output, report.output + "\n", { encoding: "utf8", flag: "wx" });
    }
    return report;
  } catch (error) {
    return { status: 2, output: "", error: "Input or output error: " + errorMessage(error) };
  }
}

// Keep the command-line entry point separate from the testable application logic.
if (process.argv[1] && import.meta.url === pathToFileURL(process.argv[1]).href) {
  const result = run(process.argv.slice(2));
  if (result.output) console.log(result.output);
  if (result.error) console.error(result.error);
  process.exitCode = result.status;
}
```

## Implementation explanation

`evaluate()` applies the recovery test rules to one record. The workflow builds its report before opening the destination in exclusive creation mode. The exported `run()` function handles arguments and returns the status and output used by the command-line entry point.

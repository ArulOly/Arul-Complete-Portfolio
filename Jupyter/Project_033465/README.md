## Complete Jupyter source code

```python
# Jupyter reference implementation for project 033465.


from IPython.display import Markdown, display
from datetime import date
from datetime import timedelta
from pathlib import Path
import json


SCENARIO = "Home Development"


def label(record, key):
    value = record.get(key)
    if not isinstance(value, str) or not value.strip():
        raise ValueError(f"{key} must be non-empty text")
    return value.strip()


def integer(record, key, minimum=0, maximum=1_000_000_000):
    value = record.get(key)
    if type(value) is not int or not minimum <= value <= maximum:
        raise ValueError(f"{key} must be an integer from {minimum} to {maximum}")
    return value


def iso_date(record, key):
    value = label(record, key)
    try:
        parsed = date.fromisoformat(value)
    except ValueError as exc:
        raise ValueError(f"{key} must use YYYY-MM-DD") from exc
    if parsed.isoformat() != value:
        raise ValueError(f"{key} must use YYYY-MM-DD")
    return parsed


def evaluate(record):
    """Calculate the next due date from an explicit reporting interval and reference date."""
    previous = iso_date(record, "previous_report")
    as_of = iso_date(record, "as_of")
    interval = integer(record, "interval_days", minimum=1, maximum=365)
    if as_of < previous:
        raise ValueError("as_of must be on or after previous_report")
    due = previous + timedelta(days=interval)
    return {"due_on": due.isoformat(), "due": as_of >= due,
            "days_overdue": max(0, (as_of - due).days)}


def as_record(value):
    if not isinstance(value, dict):
        raise ValueError("Expected an object")
    return value


def analyze(payload):
    record = as_record(payload)
    return {"case": label(record, "asset"), "scenario": SCENARIO, "result": evaluate(record)}


def run_cell(input_path):
    try:
        payload = json.loads(Path(input_path).read_text(encoding="utf-8"))
        json.dumps(payload, allow_nan=False)
        report = analyze(payload)
        def cell(value):
            # Neutralize HTML, table delimiters, and line breaks in the display.
            encoded = json.dumps(value, ensure_ascii=False)
            return "".join(f"&#{ord(char)};" if char in '\\&<>|`*_[]~' else char for char in encoded)

        lines = ["| Field | Value |", "| --- | --- |"]
        lines.extend(f"| {cell(key)} | {cell(value)} |" for key, value in report["result"].items())
        display(Markdown("\n".join(lines)))
        return report
    except (OSError, ValueError, OverflowError, RecursionError) as exc:
        print(f"Notebook input or output error: {exc}")
        return None


result = run_cell(Path("sample.json"))
```

## Implementation explanation

`evaluate()` calculates the scheduled reporting outcome. `analyze()` applies the usability improvement rules, and IPython renders the response inside the notebook.
